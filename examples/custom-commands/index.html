JavaScript 

// the Event output at the bottom of the page
const output = document.getElementById("output");

function log(message) {
    output.textContent += `${message}\n`;
    output.scrollTop = output.scrollHeight;
}

output.customCommand.define("--log-click", () => {
  log("Received --log-click");
});
let value = 0;
output.customCommand.define("--increment", () => {
  value += 1;
  count.textContent = value;
  log(`Count incremented to ${value}`);
});

output.customCommand.define("--log-source", (source) => {
  log(`Source: <${source.tagName.toLowerCase()}>`);
}, { arguments: ["source"] });

Summary Details This details Pane has three methods added to it.

show(), hide(), and toggle().


Count: 0

HTML 
<details id="custom-element-methods">
  <summary>Summary Details</summary>
</details>
<button command="show" commandfor="custom-element-methods">Show</button>
<button command="hide" commandfor="custom-element-methods">Hide</button>
<button command="toggle" commandfor="custom-element-methods">Toggle</button>
JavaScript 

const customElementMethods = document.getElementById("custom-element-methods");

customElementMethods.show = () => {
  customElementMethods.open = true;
};
customElementMethods.hide = () => {
  customElementMethods.open = false;
};
customElementMethods.toggle = () => {
  customElementMethods.open = !customElementMethods.open;
};

const textInput = document.getElementById("text-input");
textInput.customCommand.define("--set-trigger", (value) => {
  textInput.commandTrigger = value;
}, { arguments: ["source.value"] });

const textOutput = document.getElementById("text-output");
textOutput.customCommand.define("--text-display", (value) => {
  textOutput.textContent = value;
